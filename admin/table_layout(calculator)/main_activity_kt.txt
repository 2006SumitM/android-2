package com.example.practical5practice

import android.os.Bundle
import android.view.View
import android.widget.Button
import android.widget.EditText
import androidx.appcompat.app.AppCompatActivity

class MainActivity : AppCompatActivity() {

    lateinit var disp: EditText

    var num1 = 0.0
    var operator = ""

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)

        disp = findViewById(R.id.etno)
    }

    fun onButtonClick(view: View) {

        val btn = view as Button
        val text = btn.text.toString()

        when (text) {

            "C" -> {
                disp.setText("")
                num1 = 0.0
                operator = ""
            }

            "DEL" -> {
                val value = disp.text.toString()

                if (value.isNotEmpty()) {
                    disp.setText(value.dropLast(1))
                }
            }

            "+", "-", "*", "/" -> {
                num1 = disp.text.toString().toDouble()
                operator = text
                disp.setText("")
            }

            "=" -> {
                val num2 = disp.text.toString().toDouble()

                val result = when (operator) {
                    "+" -> num1 + num2
                    "-" -> num1 - num2
                    "*" -> num1 * num2
                    "/" -> num1 / num2
                    else -> 0.0
                }

                disp.setText(result.toString())
            }

            else -> {
                disp.append(text)
            }
        }
    }
}
