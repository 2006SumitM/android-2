package com.example.prac25

import android.os.Bundle
import android.view.MotionEvent
import android.widget.Button
import android.widget.EditText
import android.widget.ImageView
import android.widget.TextView
import android.widget.Toast
import androidx.activity.enableEdgeToEdge
import androidx.appcompat.app.AppCompatActivity
import androidx.core.view.ViewCompat
import androidx.core.view.WindowInsetsCompat

class MainActivity : AppCompatActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge()
        setContentView(R.layout.activity_main)
        val button = findViewById<Button>(R.id.btntxt)
        val textView = findViewById<TextView>(R.id.tvMessage)
        val editText = findViewById<EditText>(R.id.etName)
        val imge = findViewById<ImageView>(R.id.imageView)
        button.setOnClickListener {
            Toast.makeText(this, "Button click detected ", Toast.LENGTH_SHORT).show()
        }
        textView.setOnLongClickListener {
            Toast.makeText(this, "Long click detected", Toast.LENGTH_SHORT).show()
            true
        }
        editText.setOnFocusChangeListener { _, hasFocus ->
            if (hasFocus) {
                Toast.makeText(this, "Focus Received", Toast.LENGTH_SHORT).show()
            } else {
                Toast.makeText(this, "Focus lost", Toast.LENGTH_SHORT).show()
            }
            imge.setOnTouchListener { view, event ->
                when (event.actionMasked)
                {
                    MotionEvent.ACTION_DOWN ->
                        Toast.makeText(this, "Mouse down Detected", Toast.LENGTH_SHORT).show()

                    MotionEvent.ACTION_UP ->
                        Toast.makeText(this, "Mouse up Detected", Toast.LENGTH_SHORT).show()
                }
                true
            }
        }
    }
}
