package com.example.prac6
import android.os.Bundle
import android.widget.Button
import android.widget.EditText
import android.widget.TextView
import android.widget.Toast
import androidx.appcompat.app.AppCompatActivity
class MainActivity : AppCompatActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)
        val uname = findViewById<EditText>(R.id.etName)
        val pwd = findViewById<EditText>(R.id.etPassword)
        val fpwd = findViewById<TextView>(R.id.tvFP)
        val login = findViewById<Button>(R.id.btnLogin)
        val register = findViewById<Button>(R.id.btnRegister)
        register.setOnClickListener {
            Toast.makeText(
                this,
                "Visit www.college.com to register",
                Toast.LENGTH_LONG
            ).show()
        }
        fpwd.setOnClickListener {
            Toast.makeText(
                this,
                "Check your email to reset the password",
                Toast.LENGTH_LONG
            ).show()
        }
        login.setOnClickListener {

            val un = uname.text.toString().trim()
            val pw = pwd.text.toString().trim()

            if (un == "rehan" && pw == "1234") {

                Toast.makeText(
                    this,
                    "Login Successful",
                    Toast.LENGTH_LONG
                ).show()
            } else {

                Toast.makeText(
                    this,
                    "Invalid User name or password",
                    Toast.LENGTH_LONG
                ).show()
            }
        }
    }
}
