package com.example.prac22

import android.os.Bundle
import android.widget.Toast
import android.view.ContextMenu
import android.view.MenuItem
import android.view.View
import android.widget.ImageView
import android.widget.PopupMenu
import android.widget.TextView
import androidx.activity.enableEdgeToEdge
import androidx.appcompat.app.AppCompatActivity
import androidx.core.view.ViewCompat
import androidx.core.view.WindowInsetsCompat

class MainActivity : AppCompatActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)
        val img = findViewById<ImageView>(R.id.imageView)
        val tv = findViewById<TextView>(R.id.textView)
        registerForContextMenu(img)
        tv.setOnClickListener {
            val popup = PopupMenu(this,tv)
            popup.menuInflater.inflate(R.menu.popupmenu,popup.menu)
            popup.setOnMenuItemClickListener {
                item ->
                when(item.itemId){
                    R.id.mi_cut->{
                        Toast.makeText(this,"Text Selected",Toast.LENGTH_LONG).show()
                        true
                    }
                    R.id.mi_copy->{
                        Toast.makeText(this,"Text Copied to clipboard",Toast.LENGTH_LONG).show()
                        true
                    }

                    R.id.mi_paste->{
                        Toast.makeText(this,"Text Pasted",Toast.LENGTH_LONG).show()
                        true
                    }
                    else->false
                }
            }
            popup.show()
        }
    }

    override fun onCreateContextMenu(
        menu: ContextMenu?,
        v: View?,
        menuInfo: ContextMenu.ContextMenuInfo?
    ) {
        super.onCreateContextMenu(menu, v, menuInfo)
        menuInflater.inflate(R.menu.contextmenu,menu)
        menu?.setHeaderTitle("Select Option")
    }

    override fun onContextItemSelected(item: MenuItem): Boolean {
        when(item.itemId){
            R.id.mi_share->
                Toast.makeText(this, "Images Shared", Toast.LENGTH_SHORT).show()
            R.id.mi_save->
                Toast.makeText(this, "Images Save", Toast.LENGTH_SHORT).show()
        }
        return super.onContextItemSelected(item)
    }
}
